cmake_minimum_required(VERSION 3.16)
project(fp6-quickshare-app LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 5.15 REQUIRED COMPONENTS Core Gui Qml Quick DBus)
add_executable(fp6-quickshare-app main.cpp)
target_link_libraries(fp6-quickshare-app PRIVATE Qt5::Core Qt5::Gui Qt5::Qml Qt5::Quick Qt5::DBus)
target_compile_definitions(fp6-quickshare-app PRIVATE QML_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/qml")
install(TARGETS fp6-quickshare-app RUNTIME DESTINATION bin)
