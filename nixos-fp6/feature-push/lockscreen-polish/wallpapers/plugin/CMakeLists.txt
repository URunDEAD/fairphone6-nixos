cmake_minimum_required(VERSION 3.16)
project(fp6-wallpapers CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra")
find_package(Qt5 REQUIRED COMPONENTS Core Gui Qml Concurrent)
set(FP6_GREETER_DATA_DIR "/var/lib/lightdm-data" CACHE STRING "LightDM greeter data directory (fallback when XDG_GREETER_DATA_DIR is unset)")
add_library(fp6wallpapers SHARED plugin.cpp wallpapers.cpp)
target_compile_definitions(fp6wallpapers PRIVATE FP6_GREETER_DATA_DIR="${FP6_GREETER_DATA_DIR}")
target_link_libraries(fp6wallpapers Qt5::Core Qt5::Gui Qt5::Qml Qt5::Concurrent)
set(DEST "${CMAKE_INSTALL_PREFIX}/lib/lomiri-system-settings/private/Fp6/Wallpapers")
install(TARGETS fp6wallpapers LIBRARY DESTINATION ${DEST})
install(FILES qmldir DESTINATION ${DEST})
